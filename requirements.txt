# School ERP Demo - Dependencies
# ================================
# These are the Python packages needed to run the app.
#
# NOTE: the dashboard charts are hand-built SVG (see ui/charts.py), so there is
# deliberately NO plotting library here. Fewer dependencies means a faster,
# more reliable cold start on a free Hugging Face Space.

# Gradio - Creates the web UI (chat interface, dashboard)
# This is what makes your app look nice and work in a browser
# Works on both Gradio 4.x and 5.x (see build_chat_interface in app.py)
# Upper bound matters: Gradio 6.x is published, and it REMOVES the
# retry_btn/show_api arguments this app guards for, so an unbounded pin would
# break a fresh Render build rather than just warn. Render currently resolves
# 5.50.x under this constraint; app.py adapts its ChatInterface/launch kwargs
# to whichever major is installed.
gradio>=5.0,<6.0; python_version >= "3.10"

# Python 3.9 (older local machines) cannot install Gradio 5, so fall back to
# the last 4.x there. app.py supports both.
gradio>=4.44,<5.0; python_version < "3.10"

# Hugging Face Hub - Connects to free LLM (Mistral)
# This lets us use AI without paying
huggingface_hub>=0.20.0,<1.0

# Pandas - Data manipulation (tables, calculations)
# Makes working with data easier
pandas>=2.0.0,<3.0

# Python-dateutil - Date handling
# Helps with date calculations
python-dateutil>=2.8.0
